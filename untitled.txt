        visited.add(state)

import heapq

graph = {
    'A': [('B', 1), ('C', 4)],
    'B': [('D', 2), ('E', 5)],
    'C': [('E', 1)],
    'D': [('G', 3)],
    'E': [('G', 1)],
    'G': []
}
heuristic = {'A': 6, 'B': 5, 'C': 4, 'D': 3, 'E': 2, 'G': 0}

def astar(start, goal):
    heap = [(heuristic[start], 0, start, [start])]
    visited = set()
    while heap:
        f, g, node, path = heapq.heappop(heap)
        if node in visited:
            continue
        visited.add(node)
        if node == goal:
            print("Path:", path, "Cost:", g)
            return
        for nbr, w in graph[node]:
            if nbr not in visited:
                heapq.heappush(heap, (g + w + heuristic[nbr], g + w, nbr, path + [nbr]))

--- Q3) VIVA ---
Q: What is BFS?
A: Breadth First Search is an uninformed search that explores all neighbors at the
   current depth before moving to the next level, using a FIFO queue.

Q: What is the time complexity of BFS?
A: O(b^d) where b = branching factor, d = depth of goal.

Q: What is A*?
A: A* is an informed search using f(n) = g(n) + h(n), where g is actual cost from
   start and h is heuristic estimate to goal.

Q: What is a CSP?
A: Constraint Satisfaction Problem consists of variables, domains, and constraints;
   the goal is to assign values satisfying all constraints.

Q: Difference between BFS and DFS?
A: BFS uses a queue and explores level-by-level; DFS uses a stack (or recursion)
   and explores depth-first.

from collections import deque

def water_jug(capA, capB, target):
    start = (0, 0)
    queue = deque([(start, [start])])
    visited = set()

    while queue:
        (a, b), path = queue.popleft()
        if (a, b) in visited:
            continue
        visited.add((a, b))

        if a == target or b == target:
            print("Solution found!")
            for s in path:
                print(s)
            return

        next_states = [
            (capA, b), (a, capB), (0, b), (a, 0),
            (max(0, a - (capB - b)), min(capB, a + b)),
            (min(capA, a + b), max(0, b - (capA - a)))
        ]
        for ns in next_states:
            if ns not in visited:
                queue.append((ns, path + [ns]))

--- Q2) Alpha-Beta Pruning ---

def alphabeta(depth, node, is_max, values, alpha, beta, max_depth):
    if depth == max_depth:
        return values[node]

    if is_max:
        best = -float('inf')
        for i in range(2):
            val = alphabeta(depth+1, node*2+i, False, values, alpha, beta, max_depth)
            best = max(best, val)
            alpha = max(alpha, best)
            if beta <= alpha:
                break
        return best
    else:
        best = float('inf')
        for i in range(2):
            val = alphabeta(depth+1, node*2+i, True, values, alpha, beta, max_depth)
            best = min(best, val)
            beta = min(beta, best)
            if beta <= alpha:
                break
        return best

values = [3, 5, 2, 9, 0, 7]
result = alphabeta(0, 0, True, values, -float('inf'), float('inf'), 2)

def dls(graph, node, goal, limit, visited=None, path=None):
    if visited is None:
        visited = set()
        path = []
    if node == goal:
        print("Found:", path + [node])
        return True
    if limit <= 0:
        return False
    visited.add(node)
    for nbr in graph[node]:
        if nbr not in visited:
            if dls(graph, nbr, goal, limit - 1, visited, path + [node]):
                return True
    return False

--- Q2 ALT) Tic-Tac-Toe with Minimax ---

def winner(b):
    lines = b + [list(x) for x in zip(*b)] + \
            [[b[0][0],b[1][1],b[2][2]], [b[0][2],b[1][1],b[2][0]]]
    for l in lines:
        if l[0] == l[1] == l[2] and l[0] != ' ':
            return l[0]
    return None

def minimax(b, is_max):
    w = winner(b)
    if w == 'X': return 1
    if w == 'O': return -1
    if all(c != ' ' for row in b for c in row): return 0

    if is_max:
        best = -float('inf')
        for i in range(3):
            for j in range(3):
                if b[i][j] == ' ':
                    b[i][j] = 'X'
                    best = max(best, minimax(b, False))
                    b[i][j] = ' '
        return best
    else:
        best = float('inf')
        for i in range(3):
            for j in range(3):
                if b[i][j] == ' ':
                    b[i][j] = 'O'
                    best = min(best, minimax(b, True))
                    b[i][j] = ' '
        return best

board = [['X','O','X'],['O','O',' '],[' ',' ',' ']]
print("Optimal value for X:", minimax(board, True))

--- Q1) 4-Puzzle Problem using A* ---

import heapq

def h(state, goal):
    return sum(1 for i in range(4) if state[i] != goal[i] and state[i] != 0)

def neighbors(state):
    state = list(state)
    i = state.index(0)
    moves = []
    if i >= 2:
        s = state[:]; s[i], s[i-2] = s[i-2], s[i]; moves.append(tuple(s))
    if i <= 1:
        s = state[:]; s[i], s[i+2] = s[i+2], s[i]; moves.append(tuple(s))
    if i % 2 == 1:
        s = state[:]; s[i], s[i-1] = s[i-1], s[i]; moves.append(tuple(s))
    if i % 2 == 0:
        s = state[:]; s[i], s[i+1] = s[i+1], s[i]; moves.append(tuple(s))
    return moves

def astar(start, goal):
    heap = [(h(start, goal), 0, start)]
    visited = set()
    while heap:
        f, g, state = heapq.heappop(heap)
        if state in visited:
            continue
        visited.add(state)
        if state == goal:
            print("Solved in", g, "moves")
            return
        for nxt in neighbors(state):
            if nxt not in visited:
                heapq.heappush(heap, (g+1+h(nxt, goal), g+1, nxt))

astar((1, 2, 3, 0), (1, 2, 0, 3))

--- Q1) Hill Climbing ---

def f(x):
    return -x**2 + 4*x

def hill_climb(start):
    current = start
    while True:
        neighbors = [current - 0.1, current + 0.1]
        best = current
        for n in neighbors:
            if f(n) > f(best):
                best = n
        if best == current:
            break
        current = best
    return current, f(current)

x, val = hill_climb(0)
print("Max at x =", x, "value =", val)

--- Q1) Compare Minimax and Alpha-Beta Pruning ---

def minimax(depth, node, is_max, values, max_depth):
    if depth == max_depth:
        return values[node]
    if is_max:
        return max(minimax(depth+1, node*2, False, values, max_depth),
                   minimax(depth+1, node*2+1, False, values, max_depth))
    else:
        return min(minimax(depth+1, node*2, True, values, max_depth),
                   minimax(depth+1, node*2+1, True, values, max_depth))

def alphabeta(depth, node, is_max, values, alpha, beta, max_depth):
    if depth == max_depth:
        return values[node]
    if is_max:
        best = -float('inf')
        for i in range(2):
            val = alphabeta(depth+1, node*2+i, False, values, alpha, beta, max_depth)
            best = max(best, val); alpha = max(alpha, best)
            if beta <= alpha: break
        return best
    else:
        best = float('inf')
        for i in range(2):
            val = alphabeta(depth+1, node*2+i, True, values, alpha, beta, max_depth)
            best = min(best, val); beta = min(beta, best)
            if beta <= alpha: break
        return best

values = [3, 5, 2, 9, 0, 7]
print("Minimax result:", minimax(0, 0, True, values, 2))
print("Alpha-Beta result:", alphabeta(0, 0, True, values, -float('inf'), float('inf'), 2))

--- Q1) BFS Traversal for a Graph (V1..V6) ---

from collections import deque

graph = {
    'V1': ['V2', 'V3', 'V5'],
    'V2': ['V1', 'V4'],
    'V3': ['V1', 'V6'],
    'V4': ['V2', 'V5'],
    'V5': ['V1', 'V4', 'V6'],
    'V6': ['V3', 'V5']
}

def bfs(start):
    visited, q, order = set(), deque([start]), []
    while q:
        node = q.popleft()
        if node not in visited:
            visited.add(node)
            order.append(node)
            for nbr in graph[node]:
                if nbr not in visited:
                    q.append(nbr)
    return order

print("BFS:", bfs('V1'))

def dls(graph, node, goal, limit, visited=None):
    if visited is None:
        visited = set()
    if node == goal:

    return False

import heapq

goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

def h(state):
    return sum(1 for i in range(9) if state[i] != goal[i] and state[i] != 0)

def neighbors(state):
    i = state.index(0)
    res = []
    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
        r, c = divmod(i, 3)
        nr, nc = r + dr, c + dc
        if 0 <= nr < 3 and 0 <= nc < 3:
            ni = nr * 3 + nc
            lst = list(state)
            lst[i], lst[ni] = lst[ni], lst[i]
            res.append(tuple(lst))
    return res

def astar(start):
    heap = [(h(start), 0, start)]
    visited = set()
    while heap:
        f, g, state = heapq.heappop(heap)
        if state in visited: continue
        visited.add(state)
        if state == goal:
            print("Solved in", g, "moves")
            return
        for nxt in neighbors(state):
            if nxt not in visited:
                heapq.heappush(heap, (g + 1 + h(nxt), g + 1, nxt))
    print("No solution")

--- Q1) Hill Climbing ---

def f(x):
    return -x**2 + 4*x

def hill_climb(start):
    current = start
    while True:
        neighbors = [current - 0.1, current + 0.1]
        best = max(neighbors, key=f)
        if f(best) <= f(current):
            break
        current = best
    return current, f(current)

x, val = hill_climb(0)
print("Max at x =", x, "value =", val)

def dls(graph, node, goal, limit, visited=None, path=None):
    if visited is None:
        visited = set()
        path = []
    if node == goal:
        print("Found:", path + [node])
        return True
    if limit <= 0:
        return False
    visited.add(node)
    for nbr in graph[node]:
        if nbr not in visited:
            if dls(graph, nbr, goal, limit - 1, visited, path + [node]):
                return True
    return False

def ids(graph, start, goal, max_depth):
    for depth in range(max_depth + 1):
        print(f"--- Depth {depth} ---")
        if dls(graph, start, goal, depth):
            return True
    return False

def is_safe(board, row, col, n):
    for i in range(col):
        if board[row][i] == 1:
            return False
    for i, j in zip(range(row, -1, -1), range(col, -1, -1)):
        if board[i][j] == 1:
            return False
    for i, j in zip(range(row, n), range(col, -1, -1)):
        if board[i][j] == 1:
            return False
    return True

def solve(board, col, n):
    if col >= n:
        return True
    for i in range(n):
        if is_safe(board, i, col, n):
            board[i][col] = 1
            if solve(board, col + 1, n):
                return True
            board[i][col] = 0
    return False

    for row in board:
        print(row)

--- Q1) Hill Climbing for F(x) = -x^2 + 4x ---

def f(x):
    return -x**2 + 4*x

def hill_climb(start):
    current = start
    while True:
        neighbors = [current - 0.1, current + 0.1]
        best = max(neighbors, key=f)
        if f(best) <= f(current):
            break
        current = best
    return current, f(current)

x, val = hill_climb(0)
print("Max at x =", x, "value =", val)

--- Q1) A* Search (A to E) ---

import heapq

graph = {
    'A': [('B', 2), ('C', 4)],
    'B': [('D', 1), ('E', 5)],
    'C': [('D', 2)],
    'D': [('E', 3)],
    'E': []
}
heuristic = {'A': 7, 'B': 6, 'C': 3, 'D': 2, 'E': 0}

def astar(start, goal):
    heap = [(heuristic[start], 0, start, [start])]
    visited = set()
    while heap:
        f, g, node, path = heapq.heappop(heap)
        if node in visited: continue
        visited.add(node)
        if node == goal:
            print("Path:", path, "Total Cost:", g)
            return
        for nbr, w in graph[node]:
            if nbr not in visited:
                heapq.heappush(heap, (g + w + heuristic[nbr], g + w, nbr, path + [nbr]))

astar('A', 'E')

def ao_star():
    print("=== AO* Algorithm (AND-OR Graph) ===")
    # Simplified: compute cost of each node
    # OR node: min of children; AND node: sum of children
    costs = {'I': 2, 'J': 1, 'F': 2, 'G': 1}
    # D and E are OR nodes
    costs['D'] = costs['I']
    costs['E'] = costs['J']
    # B is AND node
    costs['B'] = 1 + costs['D'] + 4 + costs['E']
    # C is OR node
    costs['C'] = min(2 + costs['F'], 5 + costs['G'])
    # A is OR node
    costs['A'] = min(2 + costs['B'], 3 + costs['C'])
    print("Optimal cost from A:", costs['A'])
    print("Costs:", costs)

ao_star()


--- Q2) Minimax Algorithm ---

def minimax(depth, node, is_max, values, max_depth):
    if depth == max_depth:
        return values[node]
    if is_max:
        return max(minimax(depth+1, node*2, False, values, max_depth),
                   minimax(depth+1, node*2+1, False, values, max_depth))
    else:
        return min(minimax(depth+1, node*2, True, values, max_depth),
                   minimax(depth+1, node*2+1, True, values, max_depth))

# Tree: A(MAX) -> B(MIN), C(MIN); B -> D=3, E=5; C -> F=2, G=9
values = [3, 5, 2, 9]
print("Optimal value for MAX:", minimax(0, 0, True, values, 2))


--- Q2 ALT) Forward Chaining ---

facts = {"temperature_high", "cough"}
rules = [
    ({"temperature_high"}, "fever"),
    ({"fever", "cough"}, "flu"),
    ({"flu"}, "rest_recommended")
]

changed = True
while changed:
